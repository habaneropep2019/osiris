#include "qtetrix.h"
#include "qdragapp.h"
#include "qfont.h"

int main( int argc, char **argv )
{
    QApplication::setColorSpec( QApplication::CustomColor );
    QDragApplication a(argc,argv);
    QApplication::setFont( QFont( "helvetica", 12 ) );
    QTetrix *tetrix = new QTetrix;
    a.setMainWidget(tetrix);
    tetrix->show();
    return a.exec();
}
